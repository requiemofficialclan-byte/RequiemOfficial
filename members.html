Положи скрипт рядом с members.json") input("Нажми Enter для выхода...") exit() with open(JSON_FILE, "r", encoding="utf-8") as f: return json.load(f) def save(members): with open(JSON_FILE, "w", encoding="utf-8") as f: json.dump(members, f, ensure_ascii=False, indent=2) print(f"✅ Сохранено в {JSON_FILE}") def show_list(members): print("\n" + "─" * 60) print(f" {'#':<4} {'Ник':<24} {'Username':<22} {'Роль'}") print("─" * 75) for m in members: roles = m["role"] if isinstance(m["role"], list) else [m["role"]] roles_str = ", ".join(roles) nick = m["nick"][:22] username = m.get("username", "—")[:20] until = f" 🚫 до {m['until']}" if m.get("until") else "" print(f" {m['num']:<4} {nick:<24} @{username:<21} {roles_str}{until}") print("─" * 60) print(f" Всего: {len(members)} участников") def pick_roles(): print("\n Доступные роли:") for i, r in enumerate(ROLES, 1): print(f" {i}. {r}") print(" Введи номера через запятую (например: 3,6 = админ + наборщик)") raw = input(" Роли: ").strip() selected = [] for part in raw.split(","): part = part.strip() if part.isdigit(): idx = int(part) - 1 if 0 <= idx < len(ROLES): selected.append(ROLES[idx]) elif part in ROLES: selected.append(part) return selected if selected else ["игрок"] def renumber(members): for i, m in enumerate(members, 1): m["num"] = i def menu(): while True: members = load() show_list(members) print("\n 1. Добавить участника") print(" 2. Удалить участника") print(" 3. Изменить роль") print(" 4. Изменить ник") print(" 5. (да/нет): ").strip().lower() if confirm == "да": members = [x for x in members if x["num"] != m["num"]] renumber(members) save(members) print(f"✅ {m['nick']} удалён") else: print(" Отмена") elif choice == "3": print("\n── Изменить роль ──") num = input(" Номер участника: ").strip() found = [m for m in members if str(m["num"]) == num] if not found: print("❌ Не найден") continue m = found[0] print(f" Участник: {m['nick']}") roles = pick_roles() m["role"] = roles save(members) print(f"✅ Роль обновлена: {', '.join(roles)}") elif choice == "4": print("\n── Изменить ник ──") num = input(" Номер участника: ").strip() found = [m for m in members if str(m["num"]) == num] if not found: print("❌ Не найден") continue m = found[0] print(f" Текущий ник: {m['nick']}") new_nick = input(" Новый ник: ").strip() if new_nick: m["nick"] = new_nick save(members) print(f"✅ Ник обновлён") else: print(" Отмена") elif choice == "5": print("\n До свидания!") break else: print(" ❌ Неверный выбор") if __name__ == "__main__": print("╔══════════════════════════════════╗") print("║ Менеджер состава Requiem ║") print("╚══════════════════════════════════╝") menu()
